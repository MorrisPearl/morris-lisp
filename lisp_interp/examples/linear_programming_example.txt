# A linear programming problem for linear_programming_example.lsp:
# invest $100 million in four mortgage pools, earning as much yield as
# possible, within limits on concentration, duration, and credit risk.
#
# The variables are the dollars put in each pool. Each variable's
# coefficient in the objective is its pool's yield:
#   gnma_30      GNMA 30-year        yield 5.4%   duration 4.5 years
#   fnma_30      FNMA 30-year        yield 5.8%   duration 5.0 years
#   non_agency   non-agency prime    yield 6.6%   duration 5.5 years
#   cmo_z        CMO Z-tranche       yield 7.2%   duration 7.5 years

maximize
0.054 gnma_30 + 0.058 fnma_30 + 0.066 non_agency + 0.072 cmo_z

subject to
# invest exactly $100 million
gnma_30 + fnma_30 + non_agency + cmo_z = 100000000
# at most $40 million in any one pool
gnma_30 <= 40000000
fnma_30 <= 40000000
non_agency <= 40000000
cmo_z <= 40000000
# average duration at most 5.5 years: (sum of duration x dollars) <= 5.5 x $100 million
4.5 gnma_30 + 5 fnma_30 + 5.5 non_agency + 7.5 cmo_z <= 550000000
# at least $30 million in GNMA pools
gnma_30 >= 30000000
# at most $50 million in the two pools with credit risk
non_agency + cmo_z <= 50000000
